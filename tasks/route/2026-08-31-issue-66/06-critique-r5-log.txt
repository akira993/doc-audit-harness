   553	def render_report(template, verdict, report_date, *, reason=None, counts=None,
   554	                  history_status=None, warnings=None, sibling=None, anchor_written=False):
   555	    found = TOKEN_RE.findall(template)
   556	    if any(token not in TOKEN_COUNTS for token in found):
   557	        raise TemplateInvalid("report template contains an unknown gate token")
   558	    for token, count in TOKEN_COUNTS.items():
   559	        actual_count = found.count(token)
   560	        if ((token in OPTIONAL_TOKENS and actual_count not in (0, count))
   561	                or (token not in OPTIONAL_TOKENS and actual_count != count)):
   562	            raise TemplateInvalid(f"report template token count is invalid for {token}")
   563	    refused = verdict == "REFUSED"
   564	    values = {
   565	        "{{GATE_VERDICT}}": verdict,
   566	        "{{GATE_REASON}}": safe_json(reason if refused else "n/a"),
   567	        "{{GATE_COUNTS}}": safe_json("n/a" if refused else counts),
   568	        "{{GATE_HISTORY_STATUS}}": safe_json("n/a" if refused else history_status),
   569	        "{{GATE_WARNINGS}}": safe_json(warnings or []),
   570	        "{{GATE_SIBLING_SCAN}}": safe_json("n/a" if refused else sibling),
   571	        "{{GATE_ANCHOR_WRITTEN}}": "true" if anchor_written else "false",
   572	        "{{GATE_REPORT_DATE}}": report_date,
   573	    }
   574	    rendered = TOKEN_RE.sub(lambda match: values[match.group(0)], template)
   575	    raw = rendered.encode("utf-8")
   576	    if len(raw) > MAX_REPORT_BYTES:
   577	        raise TemplateInvalid("rendered report exceeds 4 MiB")
   578	    return raw
