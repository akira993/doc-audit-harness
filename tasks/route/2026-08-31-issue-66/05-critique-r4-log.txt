   317	def validate_phase4_contract(repo, manifest, phase4):
   318	    if not isinstance(phase4, dict):
   319	        raise Refused("phase4 evidence must be an object")
   320	    codex = phase4.get("codexReview")
   321	    if not isinstance(codex, dict):
   322	        raise Refused("codexReview evidence invalid: codexReview must be an object")
   323	    if "state" not in codex or "promptVariant" not in codex or "carryForwardSha" not in codex:
   324	        raise Refused("codexReview evidence invalid: required keys are missing")
   325	    state = codex["state"]
   326	    variant = codex["promptVariant"]
   327	    carry_sha = codex["carryForwardSha"]
   328	    if not isinstance(state, str):
   329	        raise Refused("codexReview evidence invalid: state must be a string")
   330	    if state not in CODEX_REVIEW_STATES:
   331	        raise Refused("codexReview evidence invalid: state is not recognized")
   332	    if variant not in ("full", "diff", None):
   333	        raise Refused("codexReview evidence invalid: promptVariant is not recognized")
   334	    if not isinstance(carry_sha, str):
   335	        raise Refused("codexReview evidence invalid: carryForwardSha must be a string")
   336	    key = (manifest.get("mode"), variant, state)
   337	    if key not in PHASE4_ELIGIBILITY:
   338	        raise Refused("codexReview evidence invalid: mode, promptVariant, and state conflict")
   339	    normalized = set()
   340	    unresolved = 0
   341	    findings = phase4.get("findings", [])
   342	    if not isinstance(findings, list):
   343	        raise Refused("phase findings must be an array")
   344	    for finding in findings:
   345	        if not isinstance(finding, dict) or finding.get("source") != "codex-review":
   346	            continue
   347	        value = finding.get("file")
   348	        if not isinstance(value, str) or not value:
   349	            raise Refused("codex-review finding file is missing")
   350	        severity = finding.get("severity")
   351	        if not isinstance(severity, str) or severity.upper() not in PHASE4_SEVERITIES:
   352	            raise Refused("codex-review finding severity is invalid")
   353	        try:
   354	            path = normalize_finding_path(repo, value)
   355	        except ValueError:
   356	            path = None
   357	        if path is None:
   358	            unresolved += 1
   359	        else:
   360	            normalized.add((path, severity.upper()))
   361	    rank = {"CRITICAL": 4, "HIGH": 3, "MEDIUM": 2, "LOW": 1}
   362	    ordered = [{"file": path, "severity": severity}

   364	                                            key=lambda item: (-rank[item[1]], item[0]))]
